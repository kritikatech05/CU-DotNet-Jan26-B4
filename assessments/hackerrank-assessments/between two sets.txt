 public static int getTotalX(List<int> a, List<int> b)
    {
        
        int lcm = a[0];
        for (int i = 1; i < a.Count; i++)
        {
            lcm = LCM(lcm, a[i]);
        }

        
        int gcd = b[0];
        for (int i = 1; i < b.Count; i++)
        {
            gcd = GCD(gcd, b[i]);
        }

       
        int count = 0;

        for (int x = lcm; x <= gcd; x += lcm)
        {
            if (gcd % x == 0)
            {
                count++;
            }
        }

        return count;
    }

    static int GCD(int a, int b)
    {
        while (b != 0)
        {
            int temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }

  
    static int LCM(int a, int b)
    {
        return (a / GCD(a, b)) * b;
    }
}
